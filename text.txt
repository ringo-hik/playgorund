import * as vscode from 'vscode';
import { pumbaBuild } from './pumba';
import { SwdpProvider } from './swdpTree';

export async function activate(ctx: vscode.ExtensionContext) {
  const swdpProvider = new SwdpProvider();
  ctx.subscriptions.push(
    vscode.window.registerTreeDataProvider('swdpView', swdpProvider),

    vscode.commands.registerCommand('buildAgent.pumbaBuild', () =>
      pumbaBuild(ctx)
    ),
    vscode.commands.registerCommand('buildAgent.swdpBuild', () =>
      swdpProvider.triggerManualBuild()
    ),
    vscode.commands.registerCommand('buildAgent.createTR', () =>
      swdpProvider.createTrForLastBuild()
    )
  );
}

export function deactivate() {}

import * as vscode from 'vscode';
import { createGzip } from 'zlib';
import { pipeline } from 'stream';
import fetch, { Response } from 'node-fetch';
import { createReadStream } from 'fs';
import { tmpdir } from 'os';
import { join } from 'path';
import { promises as fs } from 'fs';

export async function pumbaBuild(ctx: vscode.ExtensionContext) {
  const ws = vscode.workspace.workspaceFolders?.[0];
  if (!ws) {
    return vscode.window.showErrorMessage('워크스페이스가 없습니다.');
  }

  await vscode.window.withProgress(
    { location: vscode.ProgressLocation.Notification, title: 'PUMBA 빌드 중...' },
    async (progress) => {
      // 1) zip
      const zipPath = join(tmpdir(), `pumba_${Date.now()}.zip`);
      await vscode.commands.executeCommand('vscode.zipWorkspace', {
        rootUri: ws.uri,
        zipFile: vscode.Uri.file(zipPath)
      });

      // 2) POST /build
      const url = vscode.workspace.getConfiguration('buildAgent').get<string>('pumbaUrl') + '/build';
      const res: Response = await fetch(url, {
        method: 'POST',
        body: createReadStream(zipPath),
        headers: { 'Content-Type': 'application/zip' }
      });
      if (!res.ok) throw new Error(await res.text());

      // 3) Stream logs
      const output = vscode.window.createOutputChannel('Build Agent › Logs');
      output.show(true);
      const decoder = new TextDecoder();
      for await (const chunk of res.body) {
        output.append(decoder.decode(chunk));
      }

      await fs.unlink(zipPath);
      progress.report({ message: '완료' });
      vscode.window.showInformationMessage('PUMBA 빌드 성공');
    }
  );
}
